      第二章

      寄存器

      ✅

      • 8086cpu有14个寄存器：AX BX CX DS SI DI SP BP IP CS SS DS ES PSW。

      • 8086cpu寄存器都是16位的。

      • ABCD-X通常用来存放一般性数据，被成为通用寄存器。

      • 寄存器?X可被分为?H和?L，如AX分为AH AL，即分为AHigh ALow，前者是高八位，后者是第八位。

      • 所谓高位和数字的高位联想，都是左边是高位。

      ✅

      • 几条汇编指令：mov add，用法一样。add ax,18H; add ax,8H。

      • 数字默认16进制。

      • mov ax,bx; mov ax,7。

      • 两个操作对象位数一致，第一个位置要是变量，如mov ah,bl; mov bh, AA。

      ✅

      • 物理地址即内存在统一的逻辑存储器中的地址。

      • 内存地址空间:cpu将系统中各类存储器看作一个逻辑存储器。

      • 物理地址=段地址*16+偏移地址，通常写作 段地址:偏移地址，如cpu指令读取开始地址CS:IP。

      • 由于十六进制，故其实即 “段地址0+偏移地址”，由一个地址加法器合成。

      • 段的概念：段地址固定，偏移地址连续的内存空间。故其最大范围0000~FFFF。实践发现，0-1会自动变成FFFF，FFFF+1变成0000。

      ✅

      • 8086有四个段寄存器：CS DS SS ES。

      • CS和IP：CS叫代码 段寄存器，IP 叫指令指针寄存器，CS存段地址IP存偏移量。

      • cpu将从CS:IP单元开始读取一条指令并执行。

      • 取指执行过程：从CS:IP读取指令，该指令进入指令缓冲器；IP=IP+该指令长度，从而指向下一指令首单元；执行指令；从CS:IP读取指令…….

      • 修改CS IP命令——jmp jmp 段地址:偏移地址 段地址修改CS 偏移地址修改IP jmp “四位数” 仅修改IP jmp 某合法寄存器 仅修改IP，IP=某合法寄存器的值

      ☝实验四 程序

      编程，向内存0:2000:23F依次传送数据063(3FH)，程序中只能使用9条指令，9条指令中包括‘mov ax,4c00h’和‘int 21h’。

      在常见的x86汇编语言（如MASM、NASM）中，分号 ; 是标准的行注释符号。

      assume cs:codesg
codesg segment
    mov ax,020H		
    mov ds,ax		; 初始化ds，不能直接将数字传入段寄存器
    
    mov bx,0H		; 初始化bx因为从0
    mov cx,40H		; cx=40H即循环64次存入数据0~63
s:				    ; 标记loop跳转位置
    mov [bx],bl		; 内存0:200~0:23F是40H个字节数据,要存入字节数据，bl存入cs:bx的字节单元，因为要对应bl的大小是1字节
    inc bx			; 把bx+=1  移动偏移地址
    loop s			; 减小cx, 检查判断是否跳转
    
    mov ax,4c00H	
    int 21H			; 结束
codesg ends
end

      编写code段中的代码，将a段和b段中的数据依次相加，将结果保存到c段中。

      assume cs:code

a segment
    db 1,2,3,4,5,6,7,8  ; a 原始数据
a ends

b segment
    db 1,2,3,4,5,6,7,8  ; b 原始数据
b ends

c segment
    db 0,0,0,0,0,0,0,0  ; c 结果存储区
c ends

code segment
start: 	
	mov bx,0	; 初始化偏移，从0开始
	mov cx,8	; 设置好循环次数
    
s:	
	mov dx,a	
	mov ds,dx	; ds=a
	mov al,[bx]	; al=a[bx]
	
	mov dx,b	
	mov ds,dx	; ds=b
	add al,[bx]	; al=
	
	mov dx,c	
	mov ds,dx	; ds=c
	mov [bx],al	; c[bx]=al=a[bx]+b[bx]
	
	inc bx		;bx++
	loop s		;跳转
    
    mov ax,4c00h	
    int 21h		; 程序退出
code ends
end start	# 指明程序的开始结尾，start处设置cs=start

      ☝实验五最后一题

      编写code段代码，用push指令将a段中的前8个字型数据，逆序存储到b段。

      assume cs:code

a segment           ; 数据段a
  dw 1,2,3,4,5,6,7,8,0ah,0bh,0ch,0dh,0fh,0ffh  
a ends

b segment           ; 数据段b
  dw 0,0,0,0,0,0,0,0                  
b ends

code segment
start:
  mov ax, a       ; 赋值ds为a段的段地址
  mov ds, ax      
  
  mov ax, b      ; 赋值堆栈段ss为b段的段地址
  mov ss, ax     
  mov sp,10h	;偏移到堆栈最后的字型数据后，8个字，16个字节，0c24:0~0c24:0f，ss:sp指向0c24:10，即前者加一。
  
  mov bx, 0       ; 初始化偏移量
  mov cx, 8       ; 循环次数

s:                
  push [bx]       ; 将a段第i个字型数据存入b段倒数第i个字单元内
  add bx, 2h      ; 按字 (word) 处理数据，故+2
  loop s

  mov ax, 4c00h   ; 退出程序
  int 21h
code ends

end start

      第十章 CALL 和 RET指令

      ret、retf

      • ret —— return retf —— return far

      • ret 出栈字单元 放入IP

        相当于 pop IP

      • retf 第一次出栈 字单元 放入 IP，第二次出栈 字单元 放入CS，相当于

        pop IP

        pop CS

      • 栈里，CS总在IP底下

        | | | IP | | CS | |——-|

        结合上地址来说，CS总在IP的右边（高地址），因为sp压栈会从高到低走。

        IPCS

        ​ ↑ ​ sp

      • 出栈、压栈总是一个字单元

      • 实现从1000:0000处开始执行指令，谁知道1000:0000处有啥。

        assume cs:code
stack segment
	db 16 dup (0)
stack ends

code segment
start:	
	mov ax, stack
	mov ss, ax
	mov sp, 16
	mov ax, 1000h
	push ax
	mov ax, 00h
	push ax
	retf
code ends

      call

      标号型

      • call s (s是标号)（位移转移），将当前IP压入栈，然后近转移到s，相当于

        push IP

        jmp near ptr s(标号)

      • 1000:3处指令为call s，s处内容为s:pop ax

        执行完这两条指令后，ax数值为 3

      • call far ptr s（标号），先将CS压入栈，再将IP压栈，然后远转移到s，相当于

        push CS

        push IP

        jmp far ptr s(标号)

      寄存器型

      call 16位reg，压栈IP，IP=(reg)

      push IP

      jmp 16位reg

      内存单元型

      • call word ptr 内存单元地址，地址取出内容大小为字单元，仅压栈IP，相当于

        push IP

        jmp word ptr 如es:[bx]

      • call dword ptr 内存单元地址，地址取出内容大小为双字，先压栈CS再压栈IP，相当于

        push CS

        push IP

        jmp dword ptr 如es:[bx]
